CẤU HÌNH KẾT NỐI TRỰC TIẾP VỚI GOOGLE SHEETS

Dán đường dẫn Web App URL từ Google Apps Script của file Google Sheets nhà trường để đồng bộ hai chiều trực tuyến.

Khôi phục dữ liệu gốc từ sheet T47

Đặt lại toàn bộ số liệu 21 lớp (840 HS) và danh sách học sinh chưa hoàn thành về đúng nguyên trạng sheet T47.

Mã nguồn Google Apps Script chuẩn (Khớp 100% cột A–R, dòng 6–26 sheet T47/T48 & sheet "Danh sách HS ") 

/**
 * GOOGLE APPS SCRIPT ĐỒNG BỘ 2 CHIỀU CHO FILE "ĐỀ ÁN 06 (An sinh xã hội)"
 * Trường THCS&THPT Nà Phặc - Sở GD&ĐT Thái Nguyên
 * Cấu trúc Sheet tuần (T46, T47, T48...): Dòng 6-16 (THCS 6A..9C), Dòng 18-27 (THPT 10A..12C)
 * Cột A: Lớp | B: GVCN | C: Tổng HS | D: CCCD Đã | E: CCCD Chưa | F: ĐD2 Đã | G: ĐD2 Chưa
 * Cột H: SSK Đã | I: SSK Chưa | J: ĐD2 >=14t | K: ASXH THCS Đã tích hợp | L: ASXH THCS Chờ duyệt
 * Cột M: ASXH THCS Chưa ĐK | N: ASXH THPT Đã ĐK | O: ASXH THPT Chưa ĐK | P: CKS Đã ĐK | Q: CKS Chưa ĐK
 */

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const weekSheetName = (e && e.parameter && e.parameter.week) ? e.parameter.week : "T47";
  const sheetTuan = ss.getSheetByName(weekSheetName) || ss.getSheetByName("T47") || ss.getSheets()[0];
  const sheetDS = ss.getSheetByName("Danh sách HS ") || ss.getSheetByName("Danh sách HS");

  const values = sheetTuan.getRange("A6:Q27").getValues();
  const classes = [];

  values.forEach(row => {
    const cls = String(row[0] || "").trim();
    if (!cls || cls.toLowerCase().includes("cộng") || cls.toLowerCase().includes("tổng")) return;
    const isTHPT = /^(10|11|12)/.test(cls);
    const asxhDone = isTHPT ? (Number(row[13]) || 0) : ((Number(row[10]) || 0) + (Number(row[11]) || 0));
    const asxhPending = isTHPT ? (Number(row[14]) || 0) : (Number(row[12]) || 0);

    classes.push({
      id: cls,
      className: cls,
      level: isTHPT ? "THPT" : "THCS",
      gvcn: String(row[1] || "").trim(),
      total: Number(row[2]) || 0,
      cccdDone: Number(row[3]) || 0,
      cccdPending: Number(row[4]) || 0,
      vneid2Done: Number(row[5]) || 0,
      vneid2Pending: Number(row[6]) || 0,
      sskDone: Number(row[7]) || 0,
      sskPending: Number(row[8]) || 0,
      asxhApplicable: Number(row[9]) || 0,
      asxhDone: asxhDone,
      asxhPending: asxhPending,
      cksDone: isTHPT ? (Number(row[15]) || 0) : 0,
      cksPending: isTHPT ? (Number(row[16]) || 0) : 0,
      week: weekSheetName
    });
  });

  const students = [];
  if (sheetDS) {
    const dsLastRow = sheetDS.getLastRow();
    if (dsLastRow >= 5) {
      const dsValues = sheetDS.getRange(5, 1, dsLastRow - 4, 8).getValues();
      let autoId = 1;
      dsValues.forEach(r => {
        const name = String(r[1] || "").trim();
        const cls = String(r[2] || "").trim();
        if (!name || !cls) return;
        students.push({
          id: autoId++,
          name: name,
          className: cls,
          cccd: String(r[3] || "").trim().toUpperCase() === "X",
          vneid2: String(r[4] || "").trim().toUpperCase() === "X",
          ssk: String(r[5] || "").trim().toUpperCase() === "X",
          cks: String(r[6] || "").trim().toUpperCase() === "X",
          reason: String(r[7] || "").trim()
        });
      });
    }
  }

  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    week: sheetTuan.getName(),
    timestamp: new Date().toISOString(),
    classes: classes,
    students: students
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const body = JSON.parse(e.postData.contents);
    const c = body.classData;
    const ds = body.unfinishedStudents || [];
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Cập nhật đúng dòng của lớp trong sheet tuần (T46, T47, T48...)
    const weekSheetName = c.week || "T47";
    let sheetTuan = ss.getSheetByName(weekSheetName);
    if (!sheetTuan) {
      const template = ss.getSheetByName("T47") || ss.getSheets()[0];
      sheetTuan = template.copyTo(ss).setName(weekSheetName);
    }

    const classCol = sheetTuan.getRange("A6:A27").getValues();
    let targetRow = -1;
    for (let i = 0; i < classCol.length; i++) {
      if (String(classCol[i][0]).trim().toUpperCase() === String(c.className).trim().toUpperCase()) {
        targetRow = i + 6;
        break;
      }
    }

    if (targetRow !== -1) {
      const isTHPT = c.level === "THPT";
      // Ghi cột B..I (GVCN, Tổng HS, CCCD, ĐD2, SSK)
      sheetTuan.getRange(targetRow, 2, 1, 8).setValues([[
        c.gvcn,
        Number(c.total) || 0,
        Number(c.cccdDone) || 0,
        Number(c.cccdPending) || 0,
        Number(c.vneid2Done) || 0,
        Number(c.vneid2Pending) || 0,
        Number(c.sskDone) || 0,
        Number(c.sskPending) || 0
      ]]);

      // Ghi cột J..Q (ĐD2 >=14t, ASXH THCS/THPT, CKS)
      if (isTHPT) {
        sheetTuan.getRange(targetRow, 10).setValue(Number(c.vneid2Done) || 0);
        sheetTuan.getRange(targetRow, 14, 1, 4).setValues([[
          Number(c.asxhDone) || 0,
          Number(c.asxhPending) || 0,
          Number(c.cksDone) || 0,
          Number(c.cksPending) || 0
        ]]);
      } else {
        sheetTuan.getRange(targetRow, 10, 1, 4).setValues([[
          Number(c.asxhApplicable) || 0,
          0,
          Number(c.asxhDone) || 0,
          Number(c.asxhPending) || 0
        ]]);
      }
      sheetTuan.getRange(targetRow, 18).setValue("Đã nộp " + weekSheetName + " (" + (c.updatedAt || "") + ")");
    }

    // 2. Đồng bộ danh sách học sinh chưa hoàn thành của lớp vào sheet "Danh sách HS "
    const sheetDS = ss.getSheetByName("Danh sách HS ") || ss.getSheetByName("Danh sách HS");
    if (sheetDS) {
      const lastRow = sheetDS.getLastRow();
      const keptRows = [];
      if (lastRow >= 5) {
        const allRows = sheetDS.getRange(5, 1, lastRow - 4, 8).getValues();
        allRows.forEach(r => {
          const rowClass = String(r[2] || "").trim().toUpperCase();
          const rowName = String(r[1] || "").trim();
          if (rowName && rowClass !== String(c.className).trim().toUpperCase()) {
            keptRows.push(r);
          }
        });
      }

      ds.forEach(st => {
        keptRows.push([
          0,
          st.name,
          c.className,
          st.cccd ? "X" : "",
          st.vneid2 ? "X" : "",
          st.ssk ? "X" : "",
          st.cks ? "X" : "",
          st.reason || ""
        ]);
      });

      if (lastRow >= 5) {
        sheetDS.getRange(5, 1, lastRow - 4, 8).clearContent();
      }
      if (keptRows.length > 0) {
        const numbered = keptRows.map((r, idx) => [idx + 1, r[1], r[2], r[3], r[4], r[5], r[6], r[7]]);
        sheetDS.getRange(5, 1, numbered.length, 8).setValues(numbered);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success", className: c.className }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

Hướng dẫn triển khai 3 bước trên file Google Sheets của trường:

1. Mở file Google Sheets ĐỀ ÁN 06 (An sinh xã hội) → chọn Tiện ích mở rộng (Extensions) → Apps Script.

2. Xóa mã mặc định, dán toàn bộ đoạn mã trên vào và nhấn Lưu (Ctrl + S).

3. Chọn Triển khai (Deploy) → Tùy chọn triển khai mới (New deployment) → Loại: Ứng dụng web (Web app) → Quyền truy cập: Bất kỳ ai (Anyone) → Sao chép URL dán vào ô phía trên.
